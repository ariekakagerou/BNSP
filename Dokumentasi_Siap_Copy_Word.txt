B. DOKUMENTASI KODE PROGRAM & ARSITEKTUR SISTEM (OOP)
--------------------------------------------------------------------------------

Sistem dibangun menggunakan paradigma Pemrograman Berorientasi Objek (OOP) dengan struktur modular:

1. Class Database (classes/Database.php)
   - Fungsi: Mengelola koneksi ke database MySQL menggunakan driver PDO.
   - Penerapan OOP: Singleton Pattern & Encapsulation. Memiliki fitur Auto-Migration tabel otomatis.
   - Metode Utama: getInstance(), getConnection(), initTables().

2. Class Auth (classes/Auth.php)
   - Fungsi: Mengelola proses login autentikasi, verifikasi kata sandi (password_verify), dan session pengguna.
   - Metode Utama: login($username, $password), check(), user(), logout().

3. Class Buku (classes/Buku.php)
   - Fungsi: Mengelola operasi CRUD data buku (Create, Read, Update, Delete), pencarian kata kunci, dan filter kategori.
   - Metode Utama: getAll($search, $kategori_id), getById($id), create($data), update($id, $data), delete($id), count().

4. Class Kategori (classes/Kategori.php)
   - Fungsi: Mengelola operasi CRUD data kategori/genre buku.
   - Metode Utama: getAll(), getById($id), create($nama, $deskripsi), update($id, $nama, $deskripsi), delete($id), count().

5. Class Peminjaman (classes/Peminjaman.php)
   - Fungsi: Mengelola transaksi sirkulasi peminjaman/pengembalian buku serta otomatisasi pembaruan stok.
   - Metode Utama: getAll(), create($data), returnBook($id, $id_buku), delete($id), countActive().

1. File: classes/Database.php
--------------------------------------------------------------------------------
<?php
class Database {
    private static $instance = null;
    private $host = "localhost";
    private $db_name = "db_perpustakaan";
    private $username = "root";
    private $password = "";
    private $conn;

    private function __construct() {
        try {
            $this->conn = new PDO("mysql:host=" . $this->host . ";charset=utf8mb4", $this->username, $this->password, [
                PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
                PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC
            ]);
            $this->conn->exec("CREATE DATABASE IF NOT EXISTS `" . $this->db_name . "` DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci");
            $this->conn->exec("USE `" . $this->db_name . "`");
            $this->initTables();
        } catch (PDOException $e) {
            die("Koneksi Database Gagal: " . $e->getMessage());
        }
    }

    public static function getInstance() {
        if (self::$instance === null) {
            self::$instance = new Database();
        }
        return self::$instance;
    }

    public function getConnection() {
        return $this->conn;
    }
}
?>

2. File: classes/Buku.php
--------------------------------------------------------------------------------
<?php
require_once __DIR__ . '/Database.php';

class Buku {
    private $db;
    private $table = "buku";

    public function __construct() {
        $this->db = Database::getInstance()->getConnection();
    }

    public function getAll($search = '', $kategori_id = '') {
        $query = "SELECT b.*, k.nama_kategori FROM " . $this->table . " b LEFT JOIN kategori k ON b.id_kategori = k.id WHERE 1=1";
        if (!empty($search)) {
            $query .= " AND (b.judul LIKE :search OR b.pengarang LIKE :search OR b.isbn LIKE :search)";
        }
        if (!empty($kategori_id)) {
            $query .= " AND b.id_kategori = :kategori_id";
        }
        $query .= " ORDER BY b.id DESC";
        $stmt = $this->db->prepare($query);
        if (!empty($search)) {
            $search_param = "%" . $search . "%";
            $stmt->bindParam(':search', $search_param);
        }
        if (!empty($kategori_id)) {
            $stmt->bindParam(':kategori_id', $kategori_id);
        }
        $stmt->execute();
        return $stmt->fetchAll();
    }
}
?>
